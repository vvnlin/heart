function doGet(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { const params = e.parameter || {}; const action = params.action; const targetDateStr = params.targetDate; // 例如 2026-10-02 const name = params.name; const callback = params.callback; // JSONP 回調 // 固定綁定您的試算表 const ss = SpreadsheetApp.openById("16DSFiIll9pAbiczNoCW79EU3gjMcLzDCnHlpqhTVDAs"); let result = { success: false, message: "無動作" }; if (action === "getList") { const list = getCheckedList(ss, targetDateStr); result = { success: true, list: list }; } else if (action === "checkIn") { const monthKey = targetDateStr.substring(0, 7); const sheet = getOrCreateMonthSheet(ss, monthKey); const list = getCheckedList(ss, targetDateStr); if (!list.includes(name)) { const now = new Date(); const timeStamp = Utilities.formatDate(now, "GMT+8", "yyyy-MM-dd HH:mm:ss"); sheet.appendRow([timeStamp, targetDateStr, name, "✅ 準時合格"]); list.push(name); } result = { success: true, message: "打卡成功", list: list }; } const jsonString = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + "(" + jsonString + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonString) .setMimeType(ContentService.MimeType.JSON); } } catch (error) { const errRes = JSON.stringify({ success: false, message: error.toString() }); if (e.parameter && e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + errRes + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(errRes).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function getOrCreateMonthSheet(ss, monthKey) { let sheet = ss.getSheetByName(monthKey); if (!sheet) { sheet = ss.insertSheet(monthKey); sheet.appendRow(["打卡時間", "歸屬作業日期", "成員姓名", "繳交狀態"]); sheet.getRange(1, 1, 1, 4).setFontWeight("bold").setBackground("#f1f5f9"); sheet.setFrozenRows(1); } return sheet; } function getCheckedList(ss, targetDateStr) { if (!targetDateStr) return []; const monthKey = targetDateStr.substring(0, 7); const sheet = ss.getSheetByName(monthKey); if (!sheet) return []; const lastRow = sheet.getLastRow(); if (lastRow <= 1) return []; const data = sheet.getRange(2, 2, lastRow - 1, 2).getValues(); const checked = []; data.forEach(row => { let rowDate = row[0]; if (rowDate instanceof Date) { rowDate = Utilities.formatDate(rowDate, "GMT+8", "yyyy-MM-dd"); } if (String(rowDate) === targetDateStr && !checked.includes(row[1])) { checked.push(row[1]); } }); return checked; }
